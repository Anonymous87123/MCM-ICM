b(1)=(sum(S(k+1:2*k))-sum(S(1:k)))/k^2;

s=S-a(1);

y=a(1)+b(1)+s(1);

if i==length(S)

S(i+1)=a(end)+b(end)+s(end-k+1);

end

y(i+1)=a(i+1)+b(i+1)+s(i+1);

end
