S=reshape(X,[144,1]);

plot(S,'r');

n=length(S);

a(1)=sum(S(1:k))/k;

b(1)=(sum(S(k+1:2*k))-sum(S(1:k)))/k^2;

s=S-a(1);

y=a(1)+b(1)+s(1);

f=zeros(144,1);

if i==length(S)

S(i+1)=a(end)+b(end)+s(end-k+1);

end

a(i+1)=alpha*(S(i)-s(i))+(1-alpha)*(a(i)+b(i));

b(i+1)=beta*(a(i+1)-a(i))+(1-beta)*b(i);%

s(i+1)=gamma*(S(i)-a(i)-b(i))+(1-gamma)*s(i);%

y(i+1)=a(i+1)+b(i+1)+s(i+1);

end
