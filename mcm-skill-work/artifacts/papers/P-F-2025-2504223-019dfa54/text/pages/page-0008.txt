                                Table 2: Survival EM Algorithm

Initialize:

t←0
θt ← θ0

converged ← false

While not converged and t < maxIter do:

// E-Step: Compute Expectation of the complete-data log-likelihood
Q(θ|θt) ← 0

For i = 1 to n do:

   If δi = 1 then: // Complete data
      Q(θ|θt) ← Q(θ|θt) + log f (xi|θ)

   Else: // Censored data
      Q(θ|θt) ← Q(θ|θt) + log S(xi|θ) // Contribution from censored observa-

tion

End If

End For

// M-Step: Maximize Q(θ|θt) with respect to θ

Solve   ∂Q(θ|θt)  =  0  for  θ  to  get  θt+1
            ∂θ

// Check Convergence

If θt+1 − θt < ε or |L(θt+1) − L(θt)| < ε then:

converged ← true

End If

t←t+1
θt ← θt+1

End While
Return θt
