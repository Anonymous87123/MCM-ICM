to setup-fish
  set size 1
  set color yellow
  set shape "fish"
  set detect-radius 1.5
  set opt-temp 5.0
  set max-stay-temp 12.0
  set max-tolerable-temp 25.0
  set death-counter 0
  set fish-prop 0
  set escape-mode-counter 0
  set step 1 / 3
end

to setup
  clear-all
  reset-ticks
  initialize-variables
  setup-world
  let init-dist init-mackerel ; CHANGE FISH TO CHANGE INITIAL DISTRIBUTION
  let tot 0
  foreach [0 1 2 3 4 5] [ i ->
    set tot tot + (length item i init-dist) * (i + 1)
  ]
  foreach [0 1 2 3 4 5] [ i ->
    foreach range (length item i init-dist) [ j ->
      ask patch (item 0 item j item i init-dist) (item 1 item j item i init-dist) [
         sprout ((i + 1) / tot * (1 - rand-dist-prop) * fish-pops) [
           setup-fish
           setxy (xcor - 0.5 + random-float 1) (ycor - 0.5 + random-float 1)
         ]
      ]
    ]
  ]
  crt fish-pops - count turtles [
    setup-fish
    let test-xcor random-xcor
    let test-ycor random-ycor
    if [pcolor] of patch test-xcor test-ycor != black and [pcolor] of patch test-xcor t
      setxy test-xcor test-ycor
    ]
  ]
end
