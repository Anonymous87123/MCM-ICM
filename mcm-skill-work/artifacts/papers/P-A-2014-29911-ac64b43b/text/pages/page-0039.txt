pair<double, double> stat(vector<double> &a){
    assert(a.size()>=2);
    double mean=0;
    for(int i=0; i<a.size(); i++)
             mean+=a[i];
    mean/=a.size();
    double s=0;
    for(int i=0; i<a.size(); i++)
             s+=(a[i]-mean)*(a[i]-mean);
    s/=a.size()-1;
    s=sqrt(s);
    return make_pair(mean, s);
