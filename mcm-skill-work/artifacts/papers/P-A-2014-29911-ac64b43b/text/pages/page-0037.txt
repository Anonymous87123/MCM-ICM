int median(int *c0, int *c1){
     int cnt[7];
     for(int i=0; i<7; i++)
          cnt[i] = c0[i] + c1[i];
     int tot = 0;
     for(int i=0; i<7; i++)
          tot += cnt[i];
     tot /= 2;
     for(int i=0; tot>0 && i<7; i++){
          tot -= cnt[i];
          if (tot<=0) return i;
     }
     return 6;
}
