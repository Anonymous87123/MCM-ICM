% This a program to calculate the weight when using the AHP mathod,
% where B is the judgment matrix and W is the weight factor.
clear;clc
B=input('B=')
c=B
n=length(B);
t=sum(B);
lamada=0;
for i=1:n

   ti=t(i);
   B(:,i)=B(:,i)/ti;
end
W=sum(B,2);
W=W/sum(W);
BW=c*W;
for i=1:n

       p=BW(i)/(n*W(i));
       lamada=lamada+p;
end
CI=(lamada-n)/(n-1)
