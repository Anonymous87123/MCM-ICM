[n,m]=size(data);
k=1/log(n);
data1=data
a=max(data);
b=min(data);
%%standardization
for i=1:n

              for j=1:m
                     data(i,j)=(data(i,j)-b(j))/(a(j)-b(j))*100;

              end
end
%%weights calculating
p=zeros(n,m);e=zeros(1,m);d=zeros(1,m);
t=sum(data);
for i=1:n

       for j=1:m
              p(i,j)=data(i,j)/t(j);

       end
end
r1=p.*log(p);
r1(isnan(r1))=0;
r=sum(r1);
for j=1:m

       e(j)=-k*r(j);
       d(j)=1-e(j);
end
w=d/sum(d,2);
result=w*data1';
end
