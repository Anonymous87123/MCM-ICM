function [ output_args ] = Xovertake( m,n )
%XOVERTAKE Summary of this function goes here
%   Detailed explanation goes here
global road1;
p=find(road1(m,n+1:min(n+30,1000),1));

S=size(p);
Xr=30;
if S(2)>=1
    Xr=p(1);
end

v_ahead=road1(m,n+Xr,2);
v_me=road1(m,n,2);
if v_me>v_ahead
     b=0.9;
else
     b=0.1;
end
r=rand();
if r<b
    output_args=1;
else
     output_args=0;
end

end
