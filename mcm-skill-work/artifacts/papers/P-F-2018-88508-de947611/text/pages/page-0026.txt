Function: The code realized the k-means classification algorithm to classify all people into different groups for

              finding out the characteristics of specific people

Input:        data stored the index of each person

              N stands for the number of categories

Output: result presents the result of classification

1 function result = kmns(data,N)

2 % The function provided a method of classification

3 % according to the k-means

4 % about input :

5 % data is according to the index of each person

6 % N according to the number of categories

7 % about outout :

8 % result according to the result of Classification

9 % Initialization :

10 [m,n]=size(data);

11 result=zeros(m,n+1);

12 % Initialize the Cluster center

13 center=zeros(N,n);

14 result(:,1:n)=data(:,:);

15 while 1

16            for x=1:N

17                 % Random cluster center

18                 center(x,:)=data( randi(300,1),:);

19            end

20            distence=zeros(1,N);

21            num=zeros(1,N);

22            new_center=zeros(N,n);

23            for x=1:m

24                 for y=1:N

25                      % calculate the distance to each category

26                      distence(y)=norm(data(x,:)-center(y,:));

27                 end

28                 % calculate the minimum diatance

29                 [~, temp]=min(distence);

30                 result(x,n+1)=temp;

31            end

32            % initialization

33            k=0;

34            for y=1:N

35                 for x=1:m

36                      if result(x,n+1)==y

37                       new_center(y,:)=new_center(y,:)+result(x,1:n);

38                       num(y)=num(y)+1;

39                      end

40                 end

41                 new_center(y,:)=new_center(y,:)/num(y);

42                 if norm(new_center(y,:)-center(y,:))<0.1

43                      k=k+1;

44                 end

45            end

46            if k==N

47                  break;

48            else

49                  center=new_center;

50            end

51 end

52 [m, n]=size(result);
