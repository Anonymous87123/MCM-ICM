double gaussrand(double E, double V) {
       static double V1, V2, S;
       static int phase = 0;
       double X;
       if (phase == 0) {
              do {
                     double U1 = (double)rand() / RAND_MAX;
                     double U2 = (double)rand() / RAND_MAX;
                     V1 = 2 * U1 - 1;
                     V2 = 2 * U2 - 1;
                     S = V1 * V1 + V2 * V2;
              } while (S >= 1 || S == 0);
              X = V1 * sqrt(-2 * log(S) / S);
       }
       else X = V2 * sqrt(-2 * log(S) / S);
       phase = 1 - phase;
       return X * V + E;

}

double U_Random() {
       double f;
       f = (float)(rand() % 100);
       return f/100;

}

double possion(double Lambda) {
       double k = 0;
       long double p = 1.0;
       long double l = exp(-Lambda);
       while (p >= l) {
              double u = U_Random();
              p *= u;
              k += 1;
       }
       return k-1;

}
