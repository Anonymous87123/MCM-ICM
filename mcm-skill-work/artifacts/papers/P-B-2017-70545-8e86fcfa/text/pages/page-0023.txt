1. Efficiency simulations

   1.1 Efficiency simulation of existing model
   x=numeric(1000)
   y=numeric(1000)
   mean.ex=numeric(1000)
   PET.ex=numeric(1999)
   for (j in 1:1000)
   {
   t1=rnorm(1000,15,4)
   t2=rnorm(1000,15,4)
   for (i in 1:1000)
   {
   x[i]=sum(t1[1:i])
   y[i]=sum(t2[1:i])
   time.x.y=sort(c(x,y))
   }
   k=1:1999
   PET.ex[k]=time.x.y[k+1]-time.x.y[k]
   T.dl=sum(1-PET.ex[PET.ex¡1])
   T.dl
   mean.ex[j]=(x[1000]+T.dl)/2000
   }
   Th.ex=3600/mean.ex
   Th.ex
   mean(Th.ex)
