         %Figure 5:calculate the relationship between Rh,Rv and incidence angle

         theta=thetas./180. *pi;
         niu=10;
         lambda=1./niu;
         epsilon=80-1i.*60.*lambda.*50;
         RH1=sin(theta)-sqrt(epsilon-cos(theta).^2);
         RH2=sin(theta)+sqrt(epsilon-cos(theta).^2);
