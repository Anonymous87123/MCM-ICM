function dp_dt=population(t,p,d,z,e,c,m,max_i)
   dp_dt = zeros(max_i,1);
   for i = 1:max_i
      sum_a = 0;
      sum_b = 0;
      for j = 1:i
         sum_a = sum_a + p(j);
      end
      for j = 1:(i - 1)
         sum_b = sum_b + p(i)*c(j)*p(j);
      end
      if t-e<0
         dp_dt(i,1) = c(i)*p(i)*(1-d(t)-sum_a) - m(i)*p(i) -
            sum_b;
      else
         dp_dt(i,1) = c(i)*p(i)*(1-d(t)+z*d(t-e)-sum_a) -
            m(i)*p(i) - sum_b;
      end
   end
