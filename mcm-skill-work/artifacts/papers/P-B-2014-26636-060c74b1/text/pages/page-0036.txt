A=[0.5 0 0 1
1 0.5 0 1
1 1 0.5 1
0 0 0 0.5]

c=sum(A,2);
b=sum(A);
m=c(1,:)+b(:,1);
for i=1:4
for j=1:4
r(i,j)=(c(i,:)-c(j,:))/(2*m)+0.5;
end
end

d=sum(r);
for j=1:4
for i=1:4
s(i,j)=r(i,j)/d(:,j);
end
end

 k=power(prod(s,2),1/4);
for i=1:4
w(i,:)=k(i,:)/sum(k);
end
w=w';
