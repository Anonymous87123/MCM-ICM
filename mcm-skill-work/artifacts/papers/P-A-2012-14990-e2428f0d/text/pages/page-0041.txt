          How to calculate the area of a leaf
          l=imread('leaftype1.jpg');
          imshow(l);
          %Import the picture
          m=ginput;
          %Pick the points on the perimeter of the figure and
          record the coordinates into “m”
          x(1,:)=m(:,1);
          y(1,:)=m(:,2);
          x=[x x(1)];
          y=[y y(1)];
          N=length(x);
          t=1:N;
          ti=1:(N-1)/100000:N;
          xi=spline(t,x,ti);
          yi=spline(t,y,ti);
          area=polyarea(xi,yi);
          %calculate the area of the figure
          points=[x;y];
          fnplt(cscvn(points));axis equal
          %draw the estimated function and check the accuracy;
          save the figure as „1-1.jpg‟ for further use
