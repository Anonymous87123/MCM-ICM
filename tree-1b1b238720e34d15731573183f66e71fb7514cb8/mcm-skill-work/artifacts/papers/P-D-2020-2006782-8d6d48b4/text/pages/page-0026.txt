import numpy as np
import pandas as pd

def writeTxt(result1, result2, fileName):
       f = open(fileName,’a’)
       for ele in result1:
              f.write(str(ele)+’ ’)
       f.write(str(result2))
       f.write(’\n’)

def read_csv(filename):
       read_file = pd.read_csv(filename, header = None)
       read_file = np.array(read_file)
       return read_file

def panduan(w,re,vec_one,vec_two,m):
       flag=1
       for i in range(m):
              if re[i]==1:
                     y1=np.dot(vec_one[i],w)
                     y2=np.dot(vec_two[i],w)
                     if y1<y2:
                             flag=0
                             break
              elif re[i]==-1:
                     y1=np.dot(vec_one[i],w)
                     y2=np.dot(vec_two[i],w)
                     if y1>y2:
                             flag=0
                             break
              else:
                     y1=np.dot(vec_one[i],w)
                     y2=np.dot(vec_two[i],w)
                     if abs(y1-y2)>1:
                             flag=0
                             break
       return flag

def lowestError(w,vec_one,vec_two,m):
       y=[]
       for i in range(m):
              y1=np.dot(vec_one[i],w)
              y.append(y1)
       return np.var(y)
