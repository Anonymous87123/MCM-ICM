def predict_Mul(features, new_factor):
    predict = np.zeros([100, 1])
    for i in range(100):
        new_f = new_factor[i]
        f, w = features[i]
        pre = round(np.sum(list(map(lambda x, y: x.predict([new_f])*y, f, w))),0)
        if pre > 0:
            predict[i] = pre
        else:
            predict[i] = 0
    return predict
