Appendix: Matlab code for implementing our model

%% Geometrical Generation of Vector Tree with Bifurcations
% @para
%     theta1,phi1,theta2,phi2: specified angular parameters;
%     n: number of generations
%     start: the height of the first branching point
% output
%     x,y,z: coordinate of the branching points of a tree
function [x y z] = treeGeneration( theta1,phi1,theta2,phi2,n,start)
A=[cos(phi1) sin(phi1) 0;-cos(theta1)*sin(phi1)
cos(theta1)*cos(phi1) sin(theta1);sin(theta1)*sin(phi1)
-sin(theta1)*cos(phi1) cos(theta1)];
B=[cos(phi2) sin(phi2) 0;-cos(theta2)*sin(phi2)
cos(theta2)*cos(phi2) sin(theta2);sin(theta2)*sin(phi2)
-sin(theta2)*cos(phi2) cos(theta2)];
temp=1:2^(n);
temp=dec2bin(temp);
[row col]=size(temp);
result=zeros(3,3,row);
for p=1:row
    result(:,:,p)=eye(3,3);
end
R=zeros(3,1,row);
r=zeros(3,1,row);
r(:,:,1)=[0,0,start]';
R(:,:,1)=[0,0,start]';
for i=2:row
    temp=dec2bin(i);
    for j=2:length(temp)
        i_j=double(temp(j))-double('0');
        G_j=A^(1-i_j)*B^i_j;
        result(:,:,i)=G_j*result(:,:,i);
    end
    r(:,:,i)=result(3,:,i)';
    r(:,:,i)=normrnd(1,0.1)*r(:,:,i);
    R(:,:,i)=R(:,:,floor(i/2))+r(:,:,i);
end;
