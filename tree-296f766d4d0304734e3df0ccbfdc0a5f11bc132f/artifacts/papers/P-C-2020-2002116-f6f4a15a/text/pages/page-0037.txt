def cosdistance(v1,v2):
    vec1=v1
    vec2=v2
    fenmu1=1
    fenmu2=math.sqrt(vec2[0]*vec2[0]+vec2[1]*vec2[1]+vec2[2]*vec2[2]+vec2[3]*vec2[3
                                              ]+vec2[4]*vec2[4])
    fenzi=vec1[0]*vec2[0]+vec1[1]*vec2[1]+vec1[2]*vec2[2]+vec1[3]*vec2[3]+vec1[4]*
                                              vec2[4]
    return 1-fenzi/(fenmu1*fenmu2)
def entropy(v1):
    total=0
    vec=v1
    for i in range(5):
        total+= vec[i]*math.log(vec[i])
    return total
