#Draw the heatmap of the North Atlantic and North Sea regions.
import numpy as np
import pandas as pd
import folium
from folium.plugins import HeatMap

posi=pd.read_excel("./data/dataf2.xlsx")
num = 354
lat = np.array(posi["lat"][0:num])
lon = np.array(posi["lon"][0:num])
weight = np.array(posi["weight"][0:num],dtype=float)
data1 = [[lat[i],lon[i],weight[i]] for i in range(num)]
map_osm = folium.Map(location=[1,2],zoom_start=1)
HeatMap(data1).add_to(map_osm)
file_path = "./data/HeatMap01.html"
map_osm.save(file_path)

fish = pd.read_excel("./data/Mackerel2019.xlsx")
incidents = folium.map.FeatureGroup()
for lat, lon in zip(fish.lat, fish.lon):
   incidents.add_child(
      folium.CircleMarker(
         [lat, lon],
         radius = 7,
         color = 'yellow',
         fill = True,
         fill_color = 'red',
         fill_opacity = 0.4
      )
   )

kw1 = {
  'color': 'blue',
  'fill': True,
  'radius': 28
}
kw2 = {
  'color': 'blue',
  'fill': True,
  'radius': 20
}

san_map = folium.Map(location=[1,2], zoom_start=1)
san_map.add_child(incidents)
folium.CircleMarker([60.35, -1.2], **kw1).add_to(san_map)
folium.CircleMarker([58.73, -7.35], **kw1).add_to(san_map)
folium.CircleMarker([55.35, -9.3], **kw2).add_to(san_map)
folium.CircleMarker([57.55, -1.6], **kw2).add_to(san_map)
