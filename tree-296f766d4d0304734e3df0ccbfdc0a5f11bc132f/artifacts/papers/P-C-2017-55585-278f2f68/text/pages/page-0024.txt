Appendix A: Matlab Program for Visualization

clear all
clc
hold on;
grid on;
A=importdata(’data.txt’);
r=A(:,1);
s=A(:,2);
e=A(:,3);
f=A(:,4);
l1=A(:,5);
l2=A(:,6);
f1=f/(l1+l2)*l1/24;
f2=f/(l1+l2)*l2/24;


B=importdata(’1st_res_lane1.txt’);
d(:,1)=B(:,1);
B=importdata(’1st_res_lane2.txt’);
d(:,2)=B(:,1);
B=importdata(’1st_res_lane3.txt’);
d(:,3)=B(:,1);
B=importdata(’1st_res_lane4.txt’);
d(:,4)=B(:,1);
B=importdata(’1st_res_lane5.txt’);
d(:,5)=B(:,1);


d(:,:)=d(:,:)/(1000*1.5)*3600;
road(:)=[90 405 520];
length(:)=[25 30 13];
for mi=1:2
    for mj=1:3

        c=0;
        clear m md gca;
        for i=1:1:224
             if (r(i)==road(mj))
                 for t=s(i):0.1:e(i)
                     c=c+1;
                     m(c,1)=t;
                     if mi==1
                          m(c,2)=f1(i);
                     else
                          m(c,2)=f2(i);
                     end
                     for p=0:2:100
                          if mi==1
                               md(c,p+1)=d((100-p)/2+1,l1(i));
                          else
                               md(c,p+1)=d((100-p)/2+1,l2(i));
                          end
                     end
                 end
             end
        end
        subplot(2,3,(mi-1)*3+mj);
        hold on;

        set( gca, ’xLim’,[0, length(mj)]);
